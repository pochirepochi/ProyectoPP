CREATE VIEW dbo.vw_Usuarios_Activos AS
SELECT 
    u.Id_Usuario, u.Nombre_Usuario, u.Email, u.Nombre, u.Apellido, r.Nombre_Rol AS Rol, c.Clave_Hash, u.Estado
FROM dbo.Usuarios u
INNER JOIN dbo.Usuarios_Roles ur ON u.Id_Usuario = ur.Id_Usuario
INNER JOIN dbo.Roles r ON ur.Id_Rol = r.Id_Rol
INNER JOIN dbo.contraseña c ON u.Id_Usuario = c.id_usuario
WHERE c.es_activa = 1 AND u.Estado = 1;
GO

CREATE VIEW dbo.V_Reporte_Contactos AS
SELECT c.Razon_Social, cc.Nombre AS Contacto, cc.Telefono
FROM Clientes c INNER JOIN Contactos_Clientes cc ON c.ID = cc.FK_Cliente;
GO

CREATE VIEW dbo.V_Reporte_Direcciones AS
SELECT c.Razon_Social, CONCAT(dc.Provincia, ', ', dc.Partido, ', ', dc.Localidad) AS Direccion
FROM Clientes c INNER JOIN Direcciones_Clientes dc ON c.ID = dc.FK_Cliente;
GO

CREATE VIEW dbo.V_Reporte_Contactos_Direcciones AS
SELECT c.Razon_Social, cc.Nombre AS Contacto, cc.Telefono, CONCAT(dc.Provincia, ', ', dc.Partido, ', ', dc.Localidad) AS Direccion
FROM Clientes c LEFT JOIN Contactos_Clientes cc ON c.ID = cc.FK_Cliente LEFT JOIN Direcciones_Clientes dc ON c.ID = dc.FK_Cliente;
GO

CREATE VIEW dbo.V_Reporte_Clientes_Activos AS
SELECT ID, Razon_Social, CUIT_DNI, Limite_Credito, Activo FROM Clientes;
GO

CREATE VIEW dbo.V_Reporte_Alerta_Credito AS
SELECT 
    ID, Razon_Social, Limite_Credito,
    (ISNULL((SELECT SUM(Monto_Total) FROM Facturas_Venta WHERE FK_Cliente = Clientes.ID), 0) 
    - ISNULL((SELECT SUM(Monto) FROM Notas_Credito_Venta WHERE FK_Cliente = Clientes.ID), 0) 
    + ISNULL((SELECT SUM(Monto) FROM Notas_Debito_Venta WHERE FK_Cliente = Clientes.ID), 0)) AS Consumo_Real, Activo
FROM Clientes;
GO

CREATE VIEW dbo.V_Reporte_Historial_Notas AS
SELECT 'Crédito' AS Tipo_Nota, nc.ID AS ID_Nota, c.Razon_Social, nc.Monto, nc.Motivo, nc.Fecha, c.Activo
FROM Notas_Credito_Venta nc INNER JOIN Clientes c ON nc.FK_Cliente = c.ID
UNION ALL
SELECT 'Débito' AS Tipo_Nota, nd.ID AS ID_Nota, c.Razon_Social, nd.Monto, nd.Motivo, nd.Fecha, c.Activo
FROM Notas_Debito_Venta nd INNER JOIN Clientes c ON nd.FK_Cliente = c.ID;
GO

CREATE VIEW dbo.V_Reporte_Totales_Ajustes AS
SELECT c.ID, c.Razon_Social,
    ISNULL((SELECT SUM(Monto) FROM Notas_Credito_Venta WHERE FK_Cliente = c.ID), 0) AS Total_Acreditado,
    ISNULL((SELECT SUM(Monto) FROM Notas_Debito_Venta WHERE FK_Cliente = c.ID), 0) AS Total_Debitado
FROM Clientes c;
GO

CREATE VIEW dbo.V_Reporte_Totales_Ajustes_Detalle AS
SELECT c.ID, c.Razon_Social, 'Credito' AS Tipo, nc.Monto, nc.Fecha FROM Notas_Credito_Venta nc INNER JOIN Clientes c ON nc.FK_Cliente = c.ID
UNION ALL
SELECT c.ID, c.Razon_Social, 'Debito' AS Tipo, nd.Monto, nd.Fecha FROM Notas_Debito_Venta nd INNER JOIN Clientes c ON nd.FK_Cliente = c.ID;
GO

CREATE PROCEDURE sp_RegistrarUsuario 
    @nombre_usuario VARCHAR(50), @email VARCHAR(100), @clave VARCHAR(255), 
    @idRol INT, @nombre VARCHAR(50), @apellido VARCHAR(50), @dni VARCHAR(20), 
    @telefono VARCHAR(20), @direccion VARCHAR(255), 
    @reqLongitud BIT, @reqMayus BIT, @reqAlfanum BIT, @reqEspecial BIT
AS BEGIN
    BEGIN TRAN;
    BEGIN TRY
        INSERT INTO dbo.Usuarios (Nombre_Usuario, Email, Nombre, Apellido, DNI, Telefono, Direccion, reqLongitud, reqMayus, reqAlfanum, reqEspecial) 
        VALUES (@nombre_usuario, @email, @nombre, @apellido, @dni, @telefono, @direccion, @reqLongitud, @reqMayus, @reqAlfanum, @reqEspecial);
        
        DECLARE @IdNuevo INT = SCOPE_IDENTITY();
        INSERT INTO dbo.Usuarios_Roles (Id_Usuario, Id_Rol) VALUES (@IdNuevo, @idRol);
        INSERT INTO dbo.contraseña (id_usuario, Clave_Hash, es_activa, forzar_cambio) VALUES (@IdNuevo, @clave, 1, 0);
        COMMIT TRAN;
    END TRY
    BEGIN CATCH ROLLBACK TRAN; THROW; END CATCH
END
GO

IF OBJECT_ID('sp_ValidarUsuario', 'P') IS NOT NULL DROP PROCEDURE sp_ValidarUsuario;
GO
CREATE PROCEDURE sp_ValidarUsuario @p_nombre_usuario VARCHAR(50), @p_contrasena VARCHAR(255)
AS BEGIN
    SELECT u.Id_Usuario AS id_usuario, ur.Id_Rol AS id_Rol 
    FROM dbo.Usuarios u
    INNER JOIN dbo.Usuarios_Roles ur ON u.Id_Usuario = ur.Id_Usuario
    INNER JOIN dbo.contraseña c ON u.Id_Usuario = c.id_usuario
    WHERE u.Nombre_Usuario = @p_nombre_usuario AND c.Clave_Hash = @p_contrasena AND c.es_activa = 1 AND u.Estado = 1;
END
GO

CREATE PROCEDURE sp_ObtenerPreguntasUsuario @nombre_usuario VARCHAR(50)
AS BEGIN
    SELECT c.id_catalogo, c.Pregunta AS pregunta, r.Respuesta AS respuesta
    FROM dbo.Respuestas_Seguridad r
    INNER JOIN dbo.Catalogo_Preguntas c ON r.id_catalogo = c.id_catalogo
    INNER JOIN dbo.Usuarios u ON r.id_usuario = u.Id_Usuario
    WHERE u.Nombre_Usuario = @nombre_usuario;

IF OBJECT_ID('sp_ValidarRespuestaSeguridad', 'P') IS NOT NULL DROP PROCEDURE sp_ValidarRespuestaSeguridad;
GO
CREATE PROCEDURE sp_ValidarRespuestaSeguridad @nombre_usuario VARCHAR(50), @id_catalogo INT, @respuesta_hash VARCHAR(255), @es_valido BIT OUTPUT
AS BEGIN
    IF EXISTS (
        SELECT 1 FROM dbo.Respuestas_Seguridad r
        INNER JOIN dbo.Usuarios u ON r.id_usuario = u.Id_Usuario
        WHERE u.Nombre_Usuario = @nombre_usuario AND r.id_catalogo = @id_catalogo AND r.Respuesta = @respuesta_hash
    ) SET @es_valido = 1; ELSE SET @es_valido = 0;
END
GO

IF OBJECT_ID('sp_GuardarRespuestaSeguridad', 'P') IS NOT NULL DROP PROCEDURE sp_GuardarRespuestaSeguridad;
GO
CREATE PROCEDURE sp_GuardarRespuestaSeguridad @idUsuario INT, @idCatalogo INT, @respuestaTexto VARCHAR(255)
AS BEGIN
    IF EXISTS (SELECT 1 FROM dbo.Respuestas_Seguridad WHERE id_usuario = @idUsuario AND id_catalogo = @idCatalogo)
        UPDATE dbo.Respuestas_Seguridad SET Respuesta = @respuestaTexto WHERE id_usuario = @idUsuario AND id_catalogo = @idCatalogo;
    ELSE
        INSERT INTO dbo.Respuestas_Seguridad (id_usuario, id_catalogo, Respuesta) VALUES (@idUsuario, @idCatalogo, @respuestaTexto);
END
GO

CREATE PROCEDURE sp_Contraseñas_VerificarUltimas @id_usuario INT, @contraseña_nueva VARCHAR(255), @cantidad_a_revisar INT
AS BEGIN
    SELECT COUNT(*) FROM (
        SELECT TOP (@cantidad_a_revisar) clave_hash 
        FROM dbo.Historial_Contraseñas 
        WHERE id_usuario = @id_usuario 
        ORDER BY id_historial DESC
    ) AS Ultimas
    WHERE clave_hash = @contraseña_nueva;
END
GO

CREATE PROCEDURE sp_ModificarUsuario @idUsuario INT, @nombre VARCHAR(50), @email VARCHAR(100), @idRol INT

CREATE PROCEDURE dbo.sp_Usuario_Alta
    @Nombre_Usuario VARCHAR(50), @Email VARCHAR(100), @Nombre VARCHAR(50), @Apellido VARCHAR(50), @Id_Rol INT, @Clave_Hash VARCHAR(255), @Id_Generado INT OUTPUT
AS BEGIN
    SET NOCOUNT ON;
    SET @Nombre_Usuario = LTRIM(RTRIM(@Nombre_Usuario)); SET @Email = LTRIM(RTRIM(@Email));
    IF EXISTS (SELECT 1 FROM dbo.Usuarios WHERE Nombre_Usuario = @Nombre_Usuario) BEGIN RAISERROR('El nombre de usuario %s ya está en uso.', 16, 1, @Nombre_Usuario); RETURN; END
    IF EXISTS (SELECT 1 FROM dbo.Usuarios WHERE Email = @Email AND Estado = 1) BEGIN RAISERROR('El email %s ya está registrado.', 16, 1, @Email); RETURN; END
    IF NOT EXISTS (SELECT 1 FROM dbo.Roles WHERE Id_Rol = @Id_Rol AND Estado = 1) BEGIN RAISERROR('El Rol no existe.', 16, 1); RETURN; END
    BEGIN TRAN;
    BEGIN TRY
        INSERT INTO dbo.Usuarios (Nombre_Usuario, Email, Nombre, Apellido) VALUES (@Nombre_Usuario, @Email, @Nombre, @Apellido);
        SET @Id_Generado = SCOPE_IDENTITY();
        INSERT INTO dbo.Usuarios_Roles (Id_Usuario, Id_Rol) VALUES (@Id_Generado, @Id_Rol);
        INSERT INTO dbo.contraseña (id_usuario, Clave_Hash, es_activa, forzar_cambio) VALUES (@Id_Generado, @Clave_Hash, 1, 0);
        COMMIT TRAN;
    END TRY
    BEGIN CATCH ROLLBACK TRAN; DECLARE @Msg NVARCHAR(4000) = ERROR_MESSAGE(); RAISERROR(@Msg, 16, 1); END CATCH;
END;
GO
