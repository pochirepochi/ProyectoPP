/* ============================================================================
   1. CREACIÓN DE TABLAS Y RESTRICCIONES
   ============================================================================ */
CREATE TABLE dbo.Provincias (
    ID INT IDENTITY(1,1) NOT NULL PRIMARY KEY,
    Nombre VARCHAR(100) NULL
);
GO

CREATE TABLE dbo.Partidos (
    ID INT IDENTITY(1,1) NOT NULL PRIMARY KEY,
    FK_Provincia INT NULL FOREIGN KEY REFERENCES dbo.Provincias(ID),
    Nombre VARCHAR(100) NULL
);
GO

CREATE TABLE dbo.Localidades (
    ID INT IDENTITY(1,1) NOT NULL PRIMARY KEY,
    FK_Partido INT NULL FOREIGN KEY REFERENCES dbo.Partidos(ID),
    Nombre VARCHAR(100) NULL
);
GO

CREATE TABLE dbo.Clientes (
    ID INT IDENTITY(1,1) NOT NULL PRIMARY KEY,
    Razon_Social VARCHAR(150) NOT NULL,
    CUIT_DNI VARCHAR(20) NOT NULL UNIQUE,
    Limite_Credito DECIMAL(12, 2) NOT NULL DEFAULT ((0.00)) CHECK (Limite_Credito >= 0),
    Descuento_General DECIMAL(5, 2) NOT NULL DEFAULT ((0.00)) CHECK (Descuento_General >= 0 AND Descuento_General <= 100),
    Activo BIT NOT NULL DEFAULT ((1))
);
GO

CREATE TABLE dbo.Contactos_Clientes (
    ID INT IDENTITY(1,1) NOT NULL PRIMARY KEY,
    FK_Cliente INT NOT NULL FOREIGN KEY REFERENCES dbo.Clientes(ID) ON DELETE CASCADE ON UPDATE CASCADE,
    Nombre VARCHAR(100) NOT NULL,
    Telefono VARCHAR(50) NULL
);
GO

CREATE TABLE dbo.Direcciones_Clientes (
    ID INT IDENTITY(1,1) NOT NULL PRIMARY KEY,
    FK_Cliente INT NOT NULL FOREIGN KEY REFERENCES dbo.Clientes(ID) ON DELETE CASCADE ON UPDATE CASCADE,
    Localidad VARCHAR(100) NOT NULL,
    Provincia VARCHAR(100) NOT NULL DEFAULT (''),
    Partido VARCHAR(100) NOT NULL DEFAULT ('')
);
GO

CREATE TABLE dbo.Facturas_Venta (
    ID INT IDENTITY(1,1) NOT NULL PRIMARY KEY,
    FK_Cliente INT NOT NULL FOREIGN KEY REFERENCES dbo.Clientes(ID),
    Numero_Factura VARCHAR(50) NOT NULL UNIQUE,
    Fecha_Emision DATETIME NOT NULL DEFAULT (GETDATE()),
    Monto_Total DECIMAL(12, 2) NOT NULL
);
GO

CREATE TABLE dbo.Notas_Credito_Venta (
    ID INT IDENTITY(1,1) NOT NULL PRIMARY KEY,
    FK_Factura_Venta INT NOT NULL FOREIGN KEY REFERENCES dbo.Facturas_Venta(ID),
    FK_Cliente INT NOT NULL FOREIGN KEY REFERENCES dbo.Clientes(ID),
    Monto DECIMAL(12, 2) NOT NULL CHECK (Monto > 0),
    Motivo VARCHAR(255) NOT NULL,
    Fecha DATETIME NOT NULL DEFAULT (GETDATE())
);
GO

CREATE TABLE dbo.Notas_Debito_Venta (
    ID INT IDENTITY(1,1) NOT NULL PRIMARY KEY,
    FK_Factura_Venta INT NOT NULL FOREIGN KEY REFERENCES dbo.Facturas_Venta(ID),
    FK_Cliente INT NOT NULL FOREIGN KEY REFERENCES dbo.Clientes(ID),
    Monto DECIMAL(12, 2) NOT NULL CHECK (Monto > 0),
    Motivo VARCHAR(255) NOT NULL,
    Fecha DATETIME NOT NULL DEFAULT (GETDATE())
);
GO

CREATE TABLE dbo.Auditoria_Cambios (
    ID INT IDENTITY(1,1) NOT NULL PRIMARY KEY,
    Tabla_Afectada VARCHAR(50) NOT NULL,
    ID_Registro_Afectado INT NOT NULL,
    Accion VARCHAR(20) NOT NULL CHECK (Accion IN ('Alta', 'Modificación', 'Baja')),
    FK_Usuario INT NOT NULL,
    Fecha_Hora DATETIME NOT NULL DEFAULT (GETDATE()),
    Detalle_Cambio TEXT NULL
);
GO

/* ============================================================================
   2. VISTAS
   ============================================================================ */
CREATE VIEW dbo.V_Reporte_Contactos AS
SELECT c.Razon_Social, cc.Nombre AS Contacto, cc.Telefono
FROM Clientes c
INNER JOIN Contactos_Clientes cc ON c.ID = cc.FK_Cliente;
GO

CREATE VIEW dbo.V_Reporte_Direcciones AS
SELECT c.Razon_Social, CONCAT(dc.Provincia, ', ', dc.Partido, ', ', dc.Localidad) AS Direccion
FROM Clientes c
INNER JOIN Direcciones_Clientes dc ON c.ID = dc.FK_Cliente;
GO

CREATE VIEW dbo.V_Reporte_Totales_Ajustes_Detalle AS
SELECT c.ID, c.Razon_Social, 'Credito' AS Tipo, nc.Monto, nc.Fecha
FROM Notas_Credito_Venta nc INNER JOIN Clientes c ON nc.FK_Cliente = c.ID
UNION ALL
SELECT c.ID, c.Razon_Social, 'Debito' AS Tipo, nd.Monto, nd.Fecha
FROM Notas_Debito_Venta nd INNER JOIN Clientes c ON nd.FK_Cliente = c.ID;
GO

CREATE VIEW dbo.V_Reporte_Clientes_Activos AS
SELECT ID, Razon_Social, CUIT_DNI, Limite_Credito, Activo 
FROM Clientes;
GO

CREATE VIEW dbo.V_Reporte_Alerta_Credito AS
SELECT 
    ID, Razon_Social, Limite_Credito,
    (ISNULL((SELECT SUM(Monto_Total) FROM Facturas_Venta WHERE FK_Cliente = Clientes.ID), 0) 
    - ISNULL((SELECT SUM(Monto) FROM Notas_Credito_Venta WHERE FK_Cliente = Clientes.ID), 0) 
    + ISNULL((SELECT SUM(Monto) FROM Notas_Debito_Venta WHERE FK_Cliente = Clientes.ID), 0)) AS Consumo_Real,
    Activo
FROM Clientes;
GO

CREATE VIEW dbo.V_Reporte_Contactos_Direcciones AS
SELECT 
    c.Razon_Social,
    cc.Nombre AS Contacto,
    cc.Telefono,
    CONCAT(dc.Provincia, ', ', dc.Partido, ', ', dc.Localidad) AS Direccion
FROM Clientes c
LEFT JOIN Contactos_Clientes cc ON c.ID = cc.FK_Cliente
LEFT JOIN Direcciones_Clientes dc ON c.ID = dc.FK_Cliente;
GO

CREATE VIEW dbo.V_Reporte_Historial_Notas AS
SELECT 'Crédito' AS Tipo_Nota, nc.ID AS ID_Nota, c.Razon_Social, nc.Monto, nc.Motivo, nc.Fecha, c.Activo
FROM Notas_Credito_Venta nc
INNER JOIN Clientes c ON nc.FK_Cliente = c.ID
UNION ALL
SELECT 'Débito' AS Tipo_Nota, nd.ID AS ID_Nota, c.Razon_Social, nd.Monto, nd.Motivo, nd.Fecha, c.Activo
FROM Notas_Debito_Venta nd
INNER JOIN Clientes c ON nd.FK_Cliente = c.ID;
GO

CREATE VIEW dbo.V_Reporte_Totales_Ajustes AS
SELECT 
    c.ID,
    c.Razon_Social,
    ISNULL((SELECT SUM(Monto) FROM Notas_Credito_Venta WHERE FK_Cliente = c.ID), 0) AS Total_Acreditado,
    ISNULL((SELECT SUM(Monto) FROM Notas_Debito_Venta WHERE FK_Cliente = c.ID), 0) AS Total_Debitado
FROM Clientes c;
GO

/* ============================================================================
   3. PROCEDIMIENTOS ALMACENADOS
   ============================================================================ */
CREATE PROCEDURE dbo.SP_ActualizarCliente
    @ID_Cliente INT, @Razon_Social VARCHAR(150), @CUIT_DNI VARCHAR(20), @Limite_Credito DECIMAL(12,2), @Descuento_General DECIMAL(5,2), @Activo BIT, @FK_Usuario INT
AS BEGIN
    SET NOCOUNT ON;
    IF EXISTS (SELECT 1 FROM Clientes WHERE CUIT_DNI = @CUIT_DNI AND ID <> @ID_Cliente) BEGIN THROW 50003, 'El CUIT/DNI ingresado ya pertenece a otro cliente.', 1; END
    BEGIN TRY
        BEGIN TRANSACTION;
        UPDATE Clientes SET Razon_Social = @Razon_Social, CUIT_DNI = @CUIT_DNI, Limite_Credito = @Limite_Credito, Descuento_General = @Descuento_General, Activo = @Activo WHERE ID = @ID_Cliente;
        DECLARE @Detalle VARCHAR(255) = 'Modificación de datos del cliente ID: ' + CAST(@ID_Cliente AS VARCHAR);
        EXEC SP_RegistrarAuditoria 'Clientes', @ID_Cliente, 'Modificación', @FK_Usuario, @Detalle;
        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH ROLLBACK TRANSACTION; THROW; END CATCH
END;
GO

CREATE PROCEDURE dbo.SP_BajaCliente
    @ID_Cliente INT, @FK_Usuario INT
AS BEGIN
    SET NOCOUNT ON;
    BEGIN TRY
        BEGIN TRANSACTION;
        UPDATE Clientes SET Activo = 0 WHERE ID = @ID_Cliente;
        DECLARE @Detalle VARCHAR(255) = 'Baja lógica de cliente ID: ' + CAST(@ID_Cliente AS VARCHAR);
        EXEC SP_RegistrarAuditoria 'Clientes', @ID_Cliente, 'Baja', @FK_Usuario, @Detalle;
        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH ROLLBACK TRANSACTION; THROW; END CATCH
END;
GO

CREATE PROCEDURE dbo.SP_EmitirNotaCredito
    @FK_Factura_Venta INT, @FK_Cliente INT, @Monto DECIMAL(12,2), @Motivo VARCHAR(255), @FK_Usuario INT
AS BEGIN
    SET NOCOUNT ON;
    IF @Monto <= 0 BEGIN THROW 50004, 'El monto de la Nota de Crédito debe ser mayor a cero.', 1; END
    BEGIN TRY
        BEGIN TRANSACTION;
        INSERT INTO Notas_Credito_Venta (FK_Factura_Venta, FK_Cliente, Monto, Motivo) VALUES (@FK_Factura_Venta, @FK_Cliente, @Monto, @Motivo);
        DECLARE @NuevoID INT = SCOPE_IDENTITY();
        DECLARE @Detalle VARCHAR(255) = 'Emisión NC por monto: ' + CAST(@Monto AS VARCHAR) + ' para Cliente ID: ' + CAST(@FK_Cliente AS VARCHAR);
        EXEC SP_RegistrarAuditoria 'Notas_Credito_Venta', @NuevoID, 'Alta', @FK_Usuario, @Detalle;
        COMMIT TRANSACTION;
        SELECT @NuevoID AS ID_NotaCredito_Generada;
    END TRY
    BEGIN CATCH ROLLBACK TRANSACTION; THROW; END CATCH
END;
GO

CREATE PROCEDURE dbo.SP_EmitirNotaDebito
    @FK_Factura_Venta INT, @FK_Cliente INT, @Monto DECIMAL(12,2), @Motivo VARCHAR(255), @FK_Usuario INT
AS BEGIN
    SET NOCOUNT ON;
    BEGIN TRY
        BEGIN TRANSACTION;
        INSERT INTO Notas_Debito_Venta (FK_Factura_Venta, FK_Cliente, Monto, Motivo) VALUES (@FK_Factura_Venta, @FK_Cliente, @Monto, @Motivo);
        DECLARE @NuevoID INT = SCOPE_IDENTITY();
        DECLARE @Detalle VARCHAR(255) = 'Emisión ND por monto: ' + CAST(@Monto AS VARCHAR);
        EXEC SP_RegistrarAuditoria 'Notas_Debito_Venta', @NuevoID, 'Alta', @FK_Usuario, @Detalle;
        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH ROLLBACK TRANSACTION; THROW; END CATCH
END;
GO

CREATE PROCEDURE dbo.SP_ExisteCuitCliente
    @CUIT_DNI VARCHAR(20), @ID_Cliente_Excluir INT = 0
AS BEGIN
    SET NOCOUNT ON;
    SELECT COUNT(*) AS Cantidad FROM Clientes WHERE CUIT_DNI = @CUIT_DNI AND ID <> @ID_Cliente_Excluir;
END;
GO

CREATE PROCEDURE dbo.SP_GuardarContactoCliente
    @FK_Cliente INT, @Nombre VARCHAR(100), @Telefono VARCHAR(50)
AS BEGIN
    SET NOCOUNT ON;
    IF EXISTS (SELECT 1 FROM Contactos_Clientes WHERE FK_Cliente = @FK_Cliente)
        UPDATE Contactos_Clientes SET Nombre = @Nombre, Telefono = @Telefono WHERE FK_Cliente = @FK_Cliente;
    ELSE
        INSERT INTO Contactos_Clientes (FK_Cliente, Nombre, Telefono) VALUES (@FK_Cliente, @Nombre, @Telefono);
END;
GO

CREATE PROCEDURE dbo.SP_GuardarDireccionCliente
    @FK_Cliente INT, @Provincia VARCHAR(100), @Partido VARCHAR(100), @Localidad VARCHAR(100)
AS BEGIN
    SET NOCOUNT ON;
    IF EXISTS (SELECT 1 FROM Direcciones_Clientes WHERE FK_Cliente = @FK_Cliente)
        UPDATE Direcciones_Clientes SET Provincia = @Provincia, Partido = @Partido, Localidad = @Localidad WHERE FK_Cliente = @FK_Cliente;
    ELSE
        INSERT INTO Direcciones_Clientes (FK_Cliente, Provincia, Partido, Localidad) VALUES (@FK_Cliente, @Provincia, @Partido, @Localidad);
END;
GO

CREATE PROCEDURE dbo.SP_InsertarCliente
    @Razon_Social VARCHAR(150), @CUIT_DNI VARCHAR(20), @Limite_Credito DECIMAL(12,2), @Descuento_General DECIMAL(5,2), @Activo BIT, @FK_Usuario INT 
AS BEGIN
    SET NOCOUNT ON;
    IF EXISTS (SELECT 1 FROM Clientes WHERE CUIT_DNI = @CUIT_DNI) BEGIN THROW 50001, 'Ya existe un cliente con el CUIT/DNI ingresado.', 1; END
    IF @Limite_Credito < 0 BEGIN THROW 50002, 'El límite de crédito no puede ser negativo.', 1; END
    BEGIN TRY
        BEGIN TRANSACTION;
        INSERT INTO Clientes (Razon_Social, CUIT_DNI, Limite_Credito, Descuento_General, Activo) VALUES (@Razon_Social, @CUIT_DNI, @Limite_Credito, @Descuento_General, @Activo);
        DECLARE @NuevoID INT = SCOPE_IDENTITY();
        DECLARE @Detalle VARCHAR(255) = 'Alta de cliente CUIT: ' + @CUIT_DNI;
        EXEC SP_RegistrarAuditoria 'Clientes', @NuevoID, 'Alta', @FK_Usuario, @Detalle;
        COMMIT TRANSACTION;
        SELECT @NuevoID AS ID_Cliente_Insertado;
    END TRY
    BEGIN CATCH ROLLBACK TRANSACTION; THROW; END CATCH
END;
GO

CREATE PROCEDURE dbo.SP_ListarLocalidades AS BEGIN SELECT Nombre FROM Localidades ORDER BY Nombre; END;
GO

CREATE PROCEDURE dbo.SP_ListarLocalidadesPorPartido @Partido VARCHAR(100) AS BEGIN
    SELECT l.Nombre FROM Localidades l INNER JOIN Partidos p ON l.FK_Partido = p.ID WHERE p.Nombre = @Partido ORDER BY l.Nombre;
END;
GO

CREATE PROCEDURE dbo.SP_ListarPartidos AS BEGIN SELECT Nombre FROM Partidos ORDER BY Nombre; END;
GO

CREATE PROCEDURE dbo.SP_ListarPartidosPorProvincia @Provincia VARCHAR(100) AS BEGIN
    SELECT p.Nombre FROM Partidos p INNER JOIN Provincias prov ON p.FK_Provincia = prov.ID WHERE prov.Nombre = @Provincia ORDER BY p.Nombre;
END;
GO

CREATE PROCEDURE dbo.SP_ListarProvincias AS BEGIN SELECT Nombre FROM Provincias ORDER BY Nombre; END;
GO

CREATE PROCEDURE dbo.SP_ObtenerContactoPorCliente @FK_Cliente INT AS BEGIN
    SET NOCOUNT ON; SELECT ID, FK_Cliente, Nombre, Telefono FROM Contactos_Clientes WHERE FK_Cliente = @FK_Cliente;
END;
GO

CREATE PROCEDURE dbo.SP_ObtenerDireccionPorCliente @FK_Cliente INT AS BEGIN
    SET NOCOUNT ON; SELECT ID, FK_Cliente, Provincia, Partido, Localidad FROM Direcciones_Clientes WHERE FK_Cliente = @FK_Cliente;
END;
GO

CREATE PROCEDURE dbo.SP_RegistrarAuditoria
    @Tabla VARCHAR(50), @ID_Registro INT, @Accion VARCHAR(20), @FK_Usuario INT, @Detalle TEXT
AS BEGIN
    SET NOCOUNT ON;
    INSERT INTO Auditoria_Cambios (Tabla_Afectada, ID_Registro_Afectado, Accion, FK_Usuario, Detalle_Cambio)
    VALUES (@Tabla, @ID_Registro, @Accion, @FK_Usuario, @Detalle);
END;
GO
